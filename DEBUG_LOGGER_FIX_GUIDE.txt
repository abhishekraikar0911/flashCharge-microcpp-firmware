/**
 * DEBUG LOGGER FIX GUIDE
 * ======================
 * 
 * PROBLEM: CAN messages are printed directly with Serial.printf() 
 * instead of using the debug logger macros.
 * 
 * SOLUTION: Replace all CAN-related Serial.printf() calls with the 
 * appropriate LOG_XXX() macros.
 * 
 * MAPPING:
 * --------
 * 
 * 1. BMS Messages (CAN2 - MCP2515):
 *    - IDs: 0x1806E5F4, 0x160B8001, 0x160D8001
 *    - Replace: Serial.printf("[CAN2-RX] ...")
 *    - With: LOG_BMS("[CAN2-RX] ...")
 *    - Replace: Serial.printf("[BMS] ...")
 *    - With: LOG_BMS("[BMS] ...")
 * 
 * 2. Charger Messages (CAN1 - TWAI):
 *    - IDs: 0x0681817E, 0x0681827E, 0x00433F01, 0x00473F01
 *    - Replace: Serial.printf("[CAN1-RX] ...")
 *    - With: LOG_CHARGER("[CAN1-RX] ...")
 *    - Replace: Serial.printf("[CHARGER] ...")
 *    - With: LOG_CHARGER("[CHARGER] ...")
 * 
 * 3. OCPP Messages:
 *    - Replace: Serial.printf("[OCPP] ...")
 *    - With: LOG_OCPP("[OCPP] ...")
 *    - Replace: Serial.printf("[MO] ...")
 *    - With: LOG_OCPP("[MO] ...")
 * 
 * 4. WiFi Messages:
 *    - Replace: Serial.printf("[WiFi] ...")
 *    - With: LOG_WIFI("[WiFi] ...")
 * 
 * 5. State Machine Messages:
 *    - Replace: Serial.printf("[OCPP_SM] ...")
 *    - With: LOG_STATE("[OCPP_SM] ...")
 *    - Replace: Serial.printf("[State] ...")
 *    - With: LOG_STATE("[State] ...")
 * 
 * 6. System Messages:
 *    - Replace: Serial.printf("[System] ...")
 *    - With: LOG_SYSTEM("[System] ...")
 *    - Replace: Serial.printf("[Health] ...")
 *    - With: LOG_SYSTEM("[Health] ...")
 * 
 * EXAMPLE FIX:
 * ------------
 * 
 * // BEFORE (in can2_rx_task or BMS handler):
 * Serial.printf("[CAN2-RX] 0x%08X: ", msg.id);
 * for (int i = 0; i < msg.dlc; i++) {
 *     Serial.printf("%02X ", msg.data[i]);
 * }
 * Serial.println();
 * 
 * // AFTER:
 * char buf[128];
 * int pos = 0;
 * for (int i = 0; i < msg.dlc; i++) {
 *     pos += sprintf(buf + pos, "%02X ", msg.data[i]);
 * }
 * LOG_BMS("[CAN2-RX] 0x%08X: %s", msg.id, buf);
 * 
 * 
 * FILES TO CHECK:
 * ---------------
 * - src/main.cpp (CAN RX tasks)
 * - src/drivers/can_mcp2515_driver.cpp
 * - src/drivers/can_twai_driver.cpp
 * - src/drivers/bms_interface.cpp
 * - src/drivers/charger_interface.cpp
 * - src/modules/wifi_manager.cpp
 * - src/modules/ocpp_state_machine.cpp
 * 
 * IMPORTANT:
 * ----------
 * - Do NOT use Serial.printf() for debug messages
 * - Do NOT use Serial.println() for debug messages
 * - ONLY use LOG_BMS(), LOG_CHARGER(), LOG_OCPP(), etc.
 * - Keep Serial.printf() ONLY for:
 *   - Boot messages
 *   - Critical errors
 *   - Debug menu display
 */
